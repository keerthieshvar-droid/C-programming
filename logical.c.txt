#include<stdio.h>
int main() {
    
     int A,B;
     printf("Enter A(0 or 1):");
     scanf("%d",&A);

     printf("Enter B(0 or 1):");
     scanf("%d",&B);

     printf("\nAND = %d",A&B);
     printf("\nOR = %d",A|B);
     printf("\nXOR = %d",A^B);
     printf("\nNOT = %d",!A);
     printf("\nNOT = %d",!B);

return 0;
}
